\begin{frame}{Preprocessing and Batches}
\begin{figure}
    \centering
    \includegraphics[width=0.9\textwidth]{../results/eda/augmentation.png}
\caption{One training image and seven augmented versions of it.}
\label{fig:augmentation}
\end{figure}

\vspace{-0.2cm}
\begin{itemize}
    \item \textbf{Normalization:} $\tilde{x} = (x/255 - 0.2858) / 0.3529$, with the mean and standard deviation
          computed on the training split only.
    \item \textbf{Augmentation, training loader only:} random crop with 2-pixel edge padding, then horizontal flip
          with probability 0.5 (Figure~\ref{fig:augmentation}). Validation and test images are never augmented.
    \item \textbf{Batch:} images $(128, 1, 28, 28)$ \texttt{float32}, values in $[-0.81, 2.02]$;
          labels $(128,)$ \texttt{int64}. Every model receives this same batch.
\end{itemize}
\end{frame}

%----------------------------------------------------------
\begin{frame}{Shared Design of the Five Models}
\small
\begin{itemize}
    \item \textbf{One output contract:} every model maps a batch to raw logits of shape $(N, 10)$.
          No model applies softmax; the loss does it:
          \[
              \mathcal{L}(z, y) = -\log \frac{\exp(z_y)}{\sum_{c=1}^{10} \exp(z_c)}
          \]
    \item \textbf{Registry:} models are registered by name and built from a YAML configuration,
          so the training loop has no model-specific code.
    \item \textbf{Sequence input:} the recurrent and Transformer models receive the same image batch and convert it
          into rows, columns or patches themselves.
    \item \textbf{Self-implemented:} the five architectures, the data split, the training and evaluation loops,
          and run tracking.
    \item \textbf{From libraries:} PyTorch~\cite{paszke2019pytorch} layers (\texttt{Linear}, \texttt{Conv2d}, \texttt{GRU},
          \texttt{LSTM}, \texttt{TransformerEncoderLayer}), optimizer and scheduler; torchvision for download and transforms;
          scikit-learn for the stratified split and the metrics. No pretrained weights are used.
\end{itemize}
\end{frame}

\begin{figure}

\begin{tikzpicture}[x=1cm, y=1cm]

    \neuroncolumn{in}{0}
    \neuroncolumn{out}{2.6}
    \foreach \i in {1,...,4} \foreach \j in {1,...,4} \draw[link] (in\i) -- (out\j);
    \node[dim] at (0,-1.4) {784 inputs};

    \node[note, text=BKTemplate] at (1.3,1.35) {$z = Wx + b$};
\end{tikzpicture}
\caption{The linear classifier: one affine layer from flattened pixels to class scores.}
\label{fig:linear}
\end{figure}

\vspace{-0.2cm}
\begin{itemize}
    \item \textbf{Computation:} $x \in \mathbb{R}^{784}$, $W \in \mathbb{R}^{10 \times 784}$, $b \in \mathbb{R}^{10}$
          (Figure~\ref{fig:linear}); 7{,}850 parameters.
    \item \textbf{What it can represent:} each class score is a weighted sum of pixels, so the decision
          boundaries between classes are hyperplanes in pixel space.
    \item \textbf{Role:} the capacity floor of the comparison; every other model must improve on it.
\end{itemize}
\end{frame}

\begin{frame}{Model 2: Multilayer Perceptron (MLP)}
\begin{figure}
\centering
\begin{tikzpicture}[x=1cm, y=1cm]
    \neuroncolumn{a}{0}
    \neuroncolumn{b}{2.8}
    \neuroncolumn{c}{5.6}
    \neuroncolumn{d}{8.4}
    \foreach \i in {1,...,4} \foreach \j in {1,...,4} {
        \draw[link] (a\i) -- (b\j);
        \draw[link] (b\i) -- (c\j);
        \draw[link] (c\i) -- (d\j);
    }
    \node[dim] at (0,-1.4) {input\\784};
    \node[dim] at (2.8,-1.4) {hidden 1: 512\\ReLU, Dropout 0.2};
    \node[dim] at (5.6,-1.4) {hidden 2: 256\\ReLU, Dropout 0.2};
    \node[dim] at (8.4,-1.4) {logits\\10};
\end{tikzpicture}
\caption{The MLP: two fully connected hidden layers between the flattened image and the logits.}
\label{fig:mlp}
\end{figure}

\vspace{-0.3cm}
\small
\begin{itemize}
    \item \textbf{Computation:} $h_1 = \mathrm{Dropout}(\mathrm{ReLU}(W_1 x + b_1))$,
          $h_2 = \mathrm{Dropout}(\mathrm{ReLU}(W_2 h_1 + b_2))$, $z = W_3 h_2 + b_3$ (Figure~\ref{fig:mlp});
          535{,}818 parameters.
    \item \textbf{Activation:} $\mathrm{ReLU}(a) = \max(0, a)$ makes the model non-linear, and its gradient
          does not shrink for positive inputs.
    \item \textbf{Regularization:} dropout~\cite{srivastava2014dropout} sets 20\% of hidden units to zero during training;
          weight decay 0.05 comes from the shared optimizer.
    \item \textbf{Limit:} like the linear model, it reads a flat vector and has no notion of which pixels are neighbours.
\end{itemize}
\end{frame}

\label{fig:cnn}

\vspace{-0.3cm}
\small
\begin{itemize}
    \item \textbf{Convolution:} each $3 \times 3$ filter is applied at every position with the same weights; padding 1
          keeps the $28 \times 28$ size. One filter produces one feature map, which records where that filter responds
          (Figure~\ref{fig:cnn}).
    \item \textbf{Pooling:} $2 \times 2$ max pooling keeps the largest response in each window. It halves the map size
          and makes the output less sensitive to shifts of one pixel.
    \item \textbf{Design:} our own architecture after LeNet~\cite{lecun1998gradient}, two stages of 32 and 64 filters,
          trained from scratch.
\end{itemize}
\end{frame}

\vspace{-0.3cm}
\small
\begin{itemize}
    \item \textbf{Timestep and input size:} one timestep is one image row, read top to bottom, so the sequence has
          28 steps of 28 features (Figure~\ref{fig:gru}). Columns and $4 \times 4$ patches are the alternatives tested in E2.
    \item \textbf{Hidden representation:} $h_t = \mathrm{GRU}(x_t, h_{t-1}) \in \mathbb{R}^{128}$ summarizes the rows read so far;
          the last state $h_{28}$ is classified.
    \item \textbf{Gates:} the GRU~\cite{cho2014learning} uses an update and a reset gate to decide how much of $h_{t-1}$ to keep.
          The LSTM~\cite{hochreiter1997long} adds a separate cell state and a third gate (82{,}442 parameters); E3 compares the two.
\end{itemize}
\end{frame}

\vspace{-0.3cm}
\small
\begin{itemize}
    \item \textbf{Tokens:} each patch is flattened to 16 values and projected to $d = 128$; a learned [CLS] token is
          prepended, and its final state is classified (Figure~\ref{fig:transformer}), as in ViT~\cite{dosovitskiy2021image}.
    \item \textbf{Positional encoding:} attention ignores token order, so one learned vector per position is added to each token.
    \item \textbf{Attention~\cite{vaswani2017attention}:} queries, keys and values are linear projections of the same tokens $H$;
          each of the 4 heads computes $\mathrm{softmax}(QK^\top / \sqrt{32})\,V$, so every token can use information from
          every other token in one layer. Input and output are both $(N, 50, 128)$.
\end{itemize}
\end{frame}

\begin{frame}{The Five Models Side by Side}
\begin{table}
\centering
\caption{Input view, built-in assumption and size of each model, at the configurations in \texttt{configs/}.}
\label{tab:models}
\small
\begin{tabular}{llp{6.2cm}r}
\toprule
\textbf{Model} & \textbf{Input view} & \textbf{Built-in assumption (inductive bias)} & \textbf{Parameters} \\
\midrule
Linear      & 784 pixels           & class scores are linear in the pixels                 & 7{,}850 \\
MLP         & 784 pixels           & none about pixel layout                               & 535{,}818 \\
CNN         & $28 \times 28$ grid  & nearby pixels matter together; same filter everywhere & 421{,}642 \\
GRU         & 28 rows of 28        & the image is an ordered sequence of rows              & 62{,}218 \\
Transformer & 49 patches of 16     & few; positions are learned, all tokens interact       & 540{,}170 \\
\bottomrule
\end{tabular}
\end{table}

\small
\begin{itemize}
    \item \textbf{No spatial prior:} Linear and MLP would learn equally well if every image had its pixels shuffled by the
          same fixed permutation; the CNN and the patch Transformer would not.
    \item \textbf{Unequal size:} parameter counts differ by a factor of 69 (Table~\ref{tab:models}), so results are read
          together with cost.
\end{itemize}
\end{frame}
